package com.reconmail.crypto;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertTrue;

import java.util.Arrays;

import org.junit.jupiter.api.Test;

public class PgpCryptoLargeBodyTest {
    @Test
    public void encryptDecryptLargeBody() {
        PgpCrypto crypto = new PgpCrypto();
        KeyPairResult alice = crypto.generateKeyPair("Alice <alice@example.com>", "change-me-1234".toCharArray());
        KeyPairResult bob = crypto.generateKeyPair("Bob <bob@example.com>", "hunter2".toCharArray());

        String body = "x".repeat(10_000);
        String encrypted = crypto.encryptAndSign(body, alice.publicKeyArmored(), bob.privateKeyArmored(), "hunter2".toCharArray());

        DecryptResult decrypted = crypto.decryptAndVerify(encrypted, bob.privateKeyArmored(), "hunter2".toCharArray(), alice.publicKeyArmored());
        assertEquals(body, decrypted.plaintext());
        assertTrue(decrypted.signatureValid());
    }
}

package com.reconmail.srp;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertTrue;

import java.math.BigInteger;
import java.util.Arrays;

import org.junit.jupiter.api.Test;

public class SrpTwoLoginsTest {
    @Test
    public void twoSuccessfulLogins() {
        SrpParams params = SrpParams.default2048();
        SrpCredentials cred = SrpCredentials.newSalt();

        byte[] salt = cred.salt();
        BigInteger x = cred.deriveX(salt, "alice", "password".toCharArray());
        BigInteger v = cred.computeVerifier(params, x);

        for (int i = 0; i < 2; i++) {
            SrpClient client = new SrpClient(params, x);
            SrpServer server = new SrpServer(params, v);

            BigInteger A = client.generateA();
            BigInteger B = server.generateB();

            byte[] m1 = server.verifyClientProof(A, client.computeClientProof(B));
            assertEquals(server.serverProof(), client.verifyServerProof(B));

            // Second round
            byte[] m2 = client.computeClientProof(B);
            assertTrue(Arrays.equals(m1, m2));
        }
    }
}
